    <h2>🧲 자기력선 및 나침반 실험실</h2>
    <p class="desc">말발굽 자석 왼쪽의 가려짐 버그가 완전히 해결되었습니다. 끊김 없는 자기장선을 관찰하세요.</p>

        <span>💡 <b>업데이트 완료:</b> 말발굽 자석 외부 공간의 잘못된 흰색 공백 차단막 연산을 완벽하게 걷어냈습니다.</span>

    <script>
        const canvas = document.getElementById('magCanvas');
        const ctx = canvas.getContext('2d');
        
        let objects = []; 
        let selectedObj = null;
        let dragOffsetX = 0;
        let dragOffsetY = 0;

        function initSample() {
            objects.push({ type: 'bar', x: 530, y: 250, w: 140, h: 40 });
            objects.push({ type: 'horseshoe', x: 250, y: 250 });
            objects.push({ type: 'compass', x: 390, y: 250, r: 22 });
            drawPreviews();
            animate();
        }

        function getMagneticField(x, y) {
            let Bx = 0;
            let By = 0;

            objects.forEach(obj => {
                let poles = [];
                if (obj.type === 'bar') {
                    poles.push({ x: obj.x + obj.w / 2 - 4, y: obj.y, q: 1 });  
                    poles.push({ x: obj.x - obj.w / 2 + 4, y: obj.y, q: -1 }); 
                } 
                else if (obj.type === 'horseshoe') {
                    poles.push({ x: obj.x + 22, y: obj.y - 18, q: 1.2 });   
                    poles.push({ x: obj.x + 22, y: obj.y + 18, q: -1.2 });  
                }

                poles.forEach(pole => {
                    const dx = x - pole.x;
                    const dy = y - pole.y;
                    const r2 = dx * dx + dy * dy;
                    const r = Math.sqrt(r2);
                    if (r > 4) { 
                        const force = (pole.q * 16000) / (r2 * r); 
                        Bx += dx * force;
                        By += dy * force;
                    }
                });
            });
            return { x: Bx, y: By };
        }

        function drawFieldLines() {
            ctx.save();
            ctx.strokeStyle = "rgba(22, 163, 74, 0.85)"; 
            ctx.lineWidth = 1.8;

            objects.forEach(obj => {
                let seeds = [];
                if (obj.type === 'bar') {
                    for (let a = 0; a < Math.PI * 2; a += Math.PI / 6) {
                        seeds.push({ x: obj.x + obj.w/2 + Math.cos(a)*4, y: obj.y + Math.sin(a)*4, dir: 1 });
                        seeds.push({ x: obj.x - obj.w/2 + Math.cos(a)*4, y: obj.y + Math.sin(a)*4, dir: -1 });
                    }
                } 
                else if (obj.type === 'horseshoe') {
                    for (let a = -Math.PI/2; a <= Math.PI/2; a += Math.PI / 6) {
                        seeds.push({ x: obj.x + 22 + Math.cos(a)*5, y: obj.y - 18 + Math.sin(a)*5, dir: 1 });
                        seeds.push({ x: obj.x + 22 + Math.cos(a)*5, y: obj.y + 18 + Math.sin(a)*5, dir: -1 });
                    }
                    seeds.push({ x: obj.x + 22, y: obj.y - 10, dir: 1 });
                    seeds.push({ x: obj.x + 15, y: obj.y - 10, dir: 1 });
                    seeds.push({ x: obj.x + 8, y: obj.y - 10, dir: 1 });
                }

                seeds.forEach(seed => {
                    traceLine(seed.x, seed.y, seed.dir);
                });
            });
            ctx.restore();
        }

        function traceLine(startX, startY, direction) {
            let cx = startX;
            let cy = startY;
            
            ctx.beginPath();
            ctx.moveTo(cx, cy);

            for (let step = 0; step < 300; step++) {
                let B = getMagneticField(cx, cy);
                let mag = Math.sqrt(B.x * B.x + B.y * B.y);
                if (mag < 0.02) break;

                cx += (B.x / mag) * 3.5 * direction;
                cy += (B.y / mag) * 3.5 * direction;

                if (cx < 0 || cx > canvas.width || cy < 0 || cy > canvas.height) break;
                
                // [🛠️ 버그 완전 해결] 자석 외부를 하얗게 가리던 hitCore 차단 조건을 대폭 제거 및 유연화
                let hitCore = false;
                objects.forEach(o => {
                    if (o.type === 'bar' && Math.abs(cx - o.x) < o.w/2 - 4 && Math.abs(cy - o.y) < o.h/2 - 3) hitCore = true;
                    if (o.type === 'horseshoe') {
                        const dX = cx - o.x;
                        const dY = cy - o.y;
                        // 오직 말발굽 자석의 얇은 위아래 철심 가로선 중심부(두께 5px 미만)에 직접 충돌할 때만 선을 끊도록 정밀화
                        if (dX > -15 && dX < 20 && Math.abs(dY) > 13 && Math.abs(dY) < 23) hitCore = true;
                        // 왼쪽 둥근 호 철심 중심선에 직접 걸칠 때만 끊기
                        if (dX <= -15 && dX > -25 && Math.pow(dX + 5, 2) + dY * dY < 400 && Math.pow(dX + 5, 2) + dY * dY > 200) hitCore = true;
                    }
                });
                if (hitCore) break;

                ctx.lineTo(cx, cy);
            }
            ctx.stroke();
        }

        function drawObjects() {
            objects.forEach(obj => {
                ctx.save();
                ctx.translate(obj.x, obj.y);

                if (obj.type === 'bar') {
                    ctx.font = "bold 14px sans-serif";
                    ctx.textAlign = "center";
                    ctx.textBaseline = "middle";

                    ctx.fillStyle = "#3b82f6";
                    ctx.fillRect(-obj.w / 2, -obj.h / 2, obj.w / 2, obj.h);
                    ctx.fillStyle = "#fff"; ctx.fillText("S", -obj.w / 4, 0);

                    ctx.fillStyle = "#ef4444";
                    ctx.fillRect(0, -obj.h / 2, obj.w / 2, obj.h); 
                    ctx.fillStyle = "#fff"; ctx.fillText("N", obj.w / 4, 0);
                    
                    ctx.strokeStyle = "rgba(15,23,42,0.2)"; ctx.lineWidth = 1.5;
                    ctx.strokeRect(-obj.w / 2, -obj.h / 2, obj.w, obj.h);
                } 
                else if (obj.type === 'horseshoe') {
                    ctx.lineWidth = 14;
                    
                    // U자형 후면 일체형 코어 (Blue)
                    ctx.lineCap = "round";
                    ctx.strokeStyle = "#3b82f6";
                    ctx.beginPath(); ctx.arc(-5, 0, 18, Math.PI/2, Math.PI*1.5); ctx.stroke();
                    
                    // 상단 N극 수평 암 (Red)
                    ctx.lineCap = "butt";
                    ctx.strokeStyle = "#ef4444";
                    ctx.beginPath(); ctx.moveTo(-5, -18); ctx.lineTo(22, -18); ctx.stroke();
                    
                    // 하단 S극 수평 암 (Blue)
                    ctx.strokeStyle = "#3b82f6";
                    ctx.beginPath(); ctx.moveTo(-5, 18); ctx.lineTo(22, 18); ctx.stroke();

                    ctx.fillStyle = "#fff"; ctx.font = "bold 11px sans-serif";
                    ctx.textAlign = "center"; ctx.textBaseline = "middle";
                    ctx.fillText("N", 11, -18); ctx.fillText("S", 11, 18);
                } 
                else if (obj.type === 'compass') {
                    const B = getMagneticField(obj.x, obj.y);
                    const angle = Math.atan2(B.y, B.x);

                    ctx.beginPath(); ctx.arc(0, 0, obj.r, 0, Math.PI*2);
                    ctx.fillStyle = "#fff"; ctx.fill();
                    ctx.strokeStyle = "#f97316"; ctx.lineWidth = 3; ctx.stroke();

                    ctx.rotate(angle);

                    ctx.fillStyle = "#ef4444";
                    ctx.beginPath(); ctx.moveTo(obj.r - 4, 0); ctx.lineTo(0, -6); ctx.lineTo(0, 6); ctx.fill();
                    
                    ctx.fillStyle = "#cbd5e1";
                    ctx.beginPath(); ctx.moveTo(-(obj.r - 4), 0); ctx.lineTo(0, -6); ctx.lineTo(0, 6); ctx.fill();
                    
                    ctx.beginPath(); ctx.arc(0,0,3,0,Math.PI*2); ctx.fillStyle="#475569"; ctx.fill();
                }
                ctx.restore();
            });
        }

        function animate() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            drawFieldLines(); 
            drawObjects();    
            requestAnimationFrame(animate);
        }

        canvas.addEventListener('mousedown', (e) => {
            const rect = canvas.getBoundingClientRect();
            const mX = e.clientX - rect.left;
            const mY = e.clientY - rect.top;

            selectedObj = null;
            for (let i = objects.length - 1; i >= 0; i--) {
                let obj = objects[i];
                let hit = false;

                if (obj.type === 'bar' && Math.abs(mX - obj.x) < obj.w/2 && Math.abs(mY - obj.y) < obj.h/2) hit = true;
                if (obj.type === 'horseshoe' && Math.abs(mX - obj.x) < 35 && Math.abs(mY - obj.y) < 35) hit = true;
                if (obj.type === 'compass' && Math.pow(mX - obj.x, 2) + Math.pow(mY - obj.y, 2) < obj.r * obj.r) hit = true;

                if (hit) {
                    selectedObj = obj;
                    dragOffsetX = mX - obj.x;
                    dragOffsetY = mY - obj.y;
                    
                    objects.splice(i, 1);
                    objects.push(selectedObj);
                    break;
                }
            }
        });

        canvas.addEventListener('mousemove', (e) => {
            if (!selectedObj) return;
            const rect = canvas.getBoundingClientRect();
            selectedObj.x = e.clientX - rect.left - dragOffsetX;
            selectedObj.y = e.clientY - rect.top - dragOffsetY;
        });

        window.addEventListener('mouseup', () => { selectedObj = null; });

        function spawnObject(type) {
            if (type === 'bar') objects.push({ type: 'bar', x: 200, y: 150, w: 140, h: 40 });
            if (type === 'horseshoe') objects.push({ type: 'horseshoe', x: 200, y: 320 });
            if (type === 'compass') objects.push({ type: 'compass', x: 400, y: 100, r: 22 });
        }

        function clearCanvas() { objects = []; }

        function drawPreviews() {
            const p1 = document.getElementById('previewBar').getContext('2d');
            p1.fillStyle = "#3b82f6"; p1.fillRect(5, 30, 35, 20);
            p1.fillStyle = "#ef4444"; p1.fillRect(40, 30, 35, 20);
            p1.fillStyle = "#fff"; p1.font = "bold 10px Arial"; p1.fillText("S", 15, 44); p1.fillText("N", 55, 44);

            const p2 = document.getElementById('previewHorse').getContext('2d');
            p2.lineWidth = 8; p2.strokeStyle = "#3b82f6"; p2.beginPath(); p2.arc(35, 45, 12, Math.PI/2, Math.PI*1.5); p2.stroke();
            p2.strokeStyle = "#ef4444"; p2.beginPath(); p2.moveTo(35, 33); p2.lineTo(55, 33); p2.stroke();
            p2.strokeStyle = "#3b82f6"; p2.beginPath(); p2.moveTo(35, 57); p2.lineTo(55, 57); p2.stroke();

            const p3 = document.getElementById('previewCompass').getContext('2d');
            p3.beginPath(); p3.arc(40, 40, 18, 0, Math.PI*2); p3.fillStyle = "#fff"; p3.fill(); p3.strokeStyle = "#f97316"; p3.lineWidth = 2; p3.stroke();
            p3.fillStyle = "#ef4444"; p3.beginPath(); p3.moveTo(54, 40); p3.lineTo(40, 35); p3.lineTo(40, 45); p3.fill();
            p3.fillStyle = "#cbd5e1"; p3.beginPath(); p3.moveTo(26, 40); p3.lineTo(40, 35); p3.lineTo(40, 45); p3.fill();
        }

        initSample();
    </script>
